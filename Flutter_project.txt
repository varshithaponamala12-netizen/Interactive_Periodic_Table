class PeriodicTableApp extends StatelessWidget {
  const PeriodicTableApp({super.key});

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      title: 'Periodic Table Reference',

class PeriodicTableHome extends StatefulWidget {
  const PeriodicTableHome({super.key});

  @override
  State<PeriodicTableHome> createState() => _PeriodicTableHomeState();
}

class _PeriodicTableHomeState extends State<PeriodicTableHome> {
  String searchText = '';
  String selectedCategory = 'All';

  final List<String> categories = [
    'All',
    'Nonmetal',
    'Noble Gas',
    'Alkali Metal',
    'Alkaline Earth Metal',
    'Transition Metal',
    'Metalloid',
    'Halogen',
    'Lanthanide',
    'Actinide',
    'Post-transition Metal',
  ];

  Color categoryColor(String category) {
    switch (category) {
      case 'Nonmetal':
        return Colors.lightGreen.shade200;
      case 'Noble Gas':
        return Colors.purple.shade200;
      case 'Alkali Metal':
        return Colors.red.shade200;
      case 'Alkaline Earth Metal':
        return Colors.orange.shade200;
      case 'Transition Metal':
        return Colors.blue.shade200;
      case 'Metalloid':
        return Colors.teal.shade200;
      case 'Halogen':
        return Colors.yellow.shade300;
      case 'Lanthanide':
        return Colors.pink.shade200;
      case 'Actinide':
        return Colors.deepOrange.shade200;
      case 'Post-transition Metal':
        return Colors.cyan.shade200;
      default:
        return Colors.grey.shade300;
    }
  }

  List<Element> get filteredElements {
    final query = searchText.trim().toLowerCase();

    return elements.where((element) {
      final matchesSearch =
          element.name.toLowerCase().contains(query) ||
          element.symbol.toLowerCase().contains(query) ||
          element.number.toString() == query;

      final matchesCategory = selectedCategory == 'All' ||
          element.category == selectedCategory;

      return matchesSearch && matchesCategory;
    }).toList();
  }

  @override
  Widget build(BuildContext context) {
    final visibleNumbers =
        filteredElements.map((e) => e.number).toSet();

    return Scaffold(
      appBar: AppBar(
        title: const Text(
          'Interactive Periodic Table',
          style: TextStyle(fontWeight: FontWeight.bold),
        ),
        centerTitle: true,
        backgroundColor: Colors.indigo,
        foregroundColor: Colors.white,
      ),
      body: Column(
        children: [
          Padding(
            padding: const EdgeInsets.all(12),
            child: TextField(
              decoration: InputDecoration(
                hintText: 'Search element, symbol or atomic number',
                prefixIcon: const Icon(Icons.search),
                suffixIcon: IconButton(
                  icon: const Icon(Icons.clear),
                  onPressed: () {
                    setState(() => searchText = '');
                  },
                ),
                border: OutlineInputBorder(
                  borderRadius: BorderRadius.circular(12),
                ),
              ),
              onChanged: (value) {
                setState(() => searchText = value);
              },
            ),
          ),
          Padding(
            padding: const EdgeInsets.symmetric(horizontal: 12),
            child: DropdownButtonFormField<String>(
              value: selectedCategory,
              isExpanded: true,
              decoration: const InputDecoration(
                labelText: 'Filter by category',
                border: OutlineInputBorder(),
              ),
              items: categories.map((category) {
                return DropdownMenuItem<String>(
                  value: category,
                  child: Text(category),
                );
              }).toList(),
              onChanged: (value) {
                setState(() => selectedCategory = value ?? 'All');
              },
            ),
          ),
          Padding(
            padding: const EdgeInsets.all(10),
            child: Text(
              'Showing ${filteredElements.length} of 118 elements',
              style: const TextStyle(
                fontWeight: FontWeight.bold,
              ),
            ),
          ),
          Expanded(
            child: SingleChildScrollView(
              child: SingleChildScrollView(
                scrollDirection: Axis.horizontal,
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    const Padding(
                      padding: EdgeInsets.all(6),
                      child: Text(
                        'GROUPS  1–18',
                        style: TextStyle(
                          fontWeight: FontWeight.bold,
                          color: Colors.indigo,
                        ),
                      ),
                    ),
                    Row(
                      children: List.generate(18, (index) {
                        return SizedBox(
                          width: 61,
                          child: Center(
                            child: Text(
                              '${index + 1}',
                              style: const TextStyle(
                                fontSize: 11,
                                fontWeight: FontWeight.bold,
                              ),
                            ),
                          ),
                        );
                      }),
                    ),
                    for (int period = 0;
                        period < tableRows.length;
                        period++)
                      Row(
                        children: [
                          SizedBox(
                            width: 23,
                            child: Text(
                              '${period + 1}',
                              style: const TextStyle(
                                fontSize: 11,
                                fontWeight: FontWeight.bold,
                              ),
                            ),
                          ),
                          ...tableRows[period].map((number) {
                            if (number == 0 ||
                                !visibleNumbers.contains(number)) {
                              return const SizedBox(
                                width: 58,
                                height: 61,
                              );
                            }

                            final element = elements.firstWhere(
                              (e) => e.number == number,
                            );

                            return elementTile(element);
                          }),
                        ],
                      ),
                    const SizedBox(height: 14),
                    const Text(
                      'LANTHANIDES',
                      style: TextStyle(
                        fontWeight: FontWeight.bold,
                        color: Colors.indigo,
                      ),
                    ),
                    seriesRow(58, 71, visibleNumbers),
                    const SizedBox(height: 8),
                    const Text(
                      'ACTINIDES',
                      style: TextStyle(
                        fontWeight: FontWeight.bold,
                        color: Colors.indigo,
                      ),
                    ),
                    seriesRow(90, 103, visibleNumbers),
                    const SizedBox(height: 16),
                    const Text(
                      'Tap any element to view its details.',
                      style: TextStyle(fontStyle: FontStyle.italic),
                    ),
                    const SizedBox(height: 20),
                  ],
                ),
              ),
            ),
          ),
        ],
      ),
    );
  }

  Widget elementTile(Element element) {
    return Padding(
      padding: const EdgeInsets.all(1),
      child: InkWell(
        onTap: () {
          Navigator.push(
            context,
            MaterialPageRoute(
              builder: (_) => ElementDetailsPage(element: element),
            ),
          );
        },
        child: Container(
          width: 56,
          height: 59,
          decoration: BoxDecoration(
            color: categoryColor(element.category),
            border: Border.all(color: Colors.black26),
            borderRadius: BorderRadius.circular(5),
          ),
          child: Column(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              Text(
                '${element.number}',
                style: const TextStyle(fontSize: 9),
              ),
              Text(
                element.symbol,
                style: const TextStyle(
                  fontSize: 19,
                  fontWeight: FontWeight.bold,
                ),
              ),
              Text(
                element.name,
                style: const TextStyle(fontSize: 7),
                overflow: TextOverflow.ellipsis,
              ),
            ],
          ),
        ),
      ),
    );
  }

  Widget seriesRow(
    int start,
    int end,
    Set<int> visibleNumbers,
  ) {
    return Row(
      children: [
        for (int number = start; number <= end; number++)
          if (visibleNumbers.contains(number))
            elementTile(
              elements.firstWhere((e) => e.number == number),
            )
          else
            const SizedBox(width: 58, height: 61),
      ],
    );
  }
}

class ElementDetailsPage extends StatelessWidget {
  final Element element;

  const ElementDetailsPage({
    super.key,
    required this.element,
  });

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text(element.name),
        backgroundColor: Colors.indigo,
        foregroundColor: Colors.white,
      ),
      body: Center(
        child: SingleChildScrollView(
          padding: const EdgeInsets.all(20),
          child: Card(
            elevation: 5,
            child: Padding(
              padding: const EdgeInsets.all(24),
              child: Column(
                mainAxisSize: MainAxisSize.min,
                children: [
                  Container(
                    width: 130,
                    padding: const EdgeInsets.all(16),
                    decoration: BoxDecoration(
                      color: Colors.indigo.shade50,
                      borderRadius: BorderRadius.circular(16),
                    ),
                    child: Column(
                      children: [
                        Text(
                          '${element.number}',
                          style: const TextStyle(fontSize: 18),
                        ),
                        Text(
                          element.symbol,
                          style: const TextStyle(
                            fontSize: 60,
                            fontWeight: FontWeight.bold,
                            color: Colors.indigo,
                          ),
                        ),
                      ],
                    ),
                  ),
                  const SizedBox(height: 16),
                  Text(
                    element.name,
                    textAlign: TextAlign.center,
                    style: const TextStyle(
                      fontSize: 28,
                      fontWeight: FontWeight.bold,
                    ),
                  ),
                  const Divider(height: 32),
                  detailRow(
                    'Atomic Number',
                    '${element.number}',
                  ),
                  detailRow('Atomic Mass', element.mass),
                  detailRow('Symbol', element.symbol),
                  detailRow('Category', element.category),
                  const SizedBox(height: 20),
                  const Text(
                    'An element reference for learning '
                    'basic chemical properties.',
                    textAlign: TextAlign.center,
                  ),
                ],
              ),
            ),
          ),
        ),
      ),
    );
  }

  Widget detailRow(String title, String value) {
    return Padding(
      padding: const EdgeInsets.symmetric(vertical: 9),
      child: Row(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Expanded(
            child: Text(
              title,
              style: const TextStyle(fontWeight: FontWeight.bold),
            ),
          ),
          const SizedBox(width: 12),
          Flexible(
            child: Text(
              value,
              textAlign: TextAlign.end,
            ),
          ),
        ],
      ),
    );
  }
}
